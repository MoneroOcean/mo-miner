    set(probe "native-opencl-binding-${mode}")
    add_executable(${probe} "${MOM_SOURCE_REPO}/tests/reference/native_opencl_binding.cpp"
        "${MOM_SOURCE_REPO}/runtime/driver-library.cpp"
        "${MOM_DPCPP_SOURCE}/unified-runtime/source/adapters/opencl/ocl_dynamic_lib.cpp")

    target_include_directories(${probe} PRIVATE ${adapter_includes})
    target_compile_definitions(${probe} PRIVATE UR_STATIC_ADAPTER_OPENCL)
    if(mode STREQUAL static)
      target_compile_definitions(${probe} PRIVATE UR_STATIC_OPENCL_ICD MOM_NATIVE_OPENCL_LINKED)
    endif()
    target_link_libraries(${probe} PRIVATE OpenCL unified-runtime::headers unified-runtime::common Threads::Threads dl)
    target_link_options(${probe} PRIVATE -Wl,--wrap=dlopen -Wl,--wrap=dlerror -static-libstdc++ -static-libgcc)
